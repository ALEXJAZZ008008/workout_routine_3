% Automatically compute 40/60/80% warm-ups from editable working
% loads; always round UP to an available increment, then cap BELOW
% the working load.  Positive loads must exceed their minimum step.
\newcommand{\WarmNumber}[3]{%
  \pgfmathsetmacro{\rounded}{max(0,min((#2)-(#3),ceil((#1)*(#2)/(#3)-0.005)*(#3)))}%
  \pgfmathprintnumber[fixed,precision=1,assume math mode=true]{\rounded}}
\newcommand{\WarmupAuto}[2]{%
  \ifdim #1pt > 0pt
    {\fontsize{8.15}{10.1}\selectfont\color{muted}%
      \mbox{6\,\texttimes\,\WarmNumber{0.4}{#1}{#2}\hspace{2mm}%
      4\,\texttimes\,\WarmNumber{0.6}{#1}{#2}\hspace{2mm}%
      2\,\texttimes\,\WarmNumber{0.8}{#1}{#2}\,kg}}%
  \else
    {\fontsize{8.15}{10.1}\selectfont\color{muted}---}%
  \fi}

% For later upper-body movements, use one preparatory set of four
% repetitions at 60% of the working load (rounded up, below work load).
\newcommand{\WarmupSixtyAuto}[2]{%
  \ifdim #1pt > 0pt
    {\fontsize{8.15}{10.1}\selectfont\color{muted}%
      \mbox{4\,\texttimes\,\WarmNumber{0.6}{#1}{#2}\,kg}}%
  \else
    {\fontsize{8.15}{10.1}\selectfont\color{muted}---}%
  \fi}

{\fontsize{9.6}{11.2}\selectfont\bfseries PROGRESSION \& RECOVERY}\par\vspace{0.65mm}
\begin{minipage}[t]{0.48\linewidth}
{\fontsize{8.45}{10.2}\selectfont
\textbf{Reps:} Add one total rep to Sets 1--2 per successful session (8/8, 9/8, 9/9). In 3-set mode, Set 3 copies Set 2; repeat missed targets.\par\vspace{0.45mm}
\textbf{Weight:} Increase only after all sets reach the range maximum, then restart at its minimum. If too heavy, record actual reps and rebuild.\par\vspace{0.45mm}
\textbf{Effort:} Aim for about 1 RIR; final set may reach 0--1 RIR. Technique takes priority.
}
\end{minipage}\hfill%
\begin{minipage}[t]{0.48\linewidth}
{\fontsize{8.45}{10.2}\selectfont
\textbf{Rest:} Alternate sets, resting only while your partner works. Equipment changes provide extra recovery. Pause if needed for safe technique.\par\vspace{0.45mm}
\textbf{Plateaus:} Off days and isolated stalls are normal. Better control, technique or range of motion (including fuller partials) also count as progress. If at least half of movements stall or regress across 3 consecutive sessions, skip the next session for recovery, then resume.
}
\end{minipage}
